cmake_minimum_required(VERSION 3.20)
project(googletest_discovery LANGUAGES NONE)

# Fake providers exercise dependency selection without downloading or compiling GoogleTest.
set(package_dir "${CMAKE_CURRENT_BINARY_DIR}/package")
set(source_dir "${CMAKE_CURRENT_BINARY_DIR}/source")
file(CONFIGURE OUTPUT "${package_dir}/GTestConfig.cmake" CONTENT [[
if(NOT CPPJIEBA_TEST_GTEST_PROVIDER STREQUAL "installed")
    message(FATAL_ERROR "An existing GoogleTest target should bypass package discovery")
endif()
add_library(GTest::gtest INTERFACE IMPORTED)
add_library(GTest::gtest_main INTERFACE IMPORTED)
set_property(TARGET GTest::gtest PROPERTY CPPJIEBA_TEST_PROVIDER installed)
]] @ONLY)
file(CONFIGURE OUTPUT "${source_dir}/CMakeLists.txt" CONTENT [[
cmake_minimum_required(VERSION 3.20)
project(googletest_fixture LANGUAGES NONE)
if(NOT CPPJIEBA_TEST_GTEST_PROVIDER STREQUAL "source")
    message(FATAL_ERROR "An available GoogleTest target or package should bypass FetchContent")
endif()
add_library(gtest INTERFACE)
add_library(GTest::gtest ALIAS gtest)
set_property(TARGET gtest PROPERTY CPPJIEBA_TEST_PROVIDER source)
]] @ONLY)
set(GTest_DIR "${package_dir}" CACHE PATH "" FORCE)
set(FETCHCONTENT_SOURCE_DIR_GOOGLETEST "${source_dir}" CACHE PATH "" FORCE)

if(CPPJIEBA_TEST_GTEST_PROVIDER STREQUAL "parent")
    add_library(GTest::gtest INTERFACE IMPORTED)
    set_property(TARGET GTest::gtest PROPERTY CPPJIEBA_TEST_PROVIDER parent)
elseif(CPPJIEBA_TEST_GTEST_PROVIDER STREQUAL "source")
    set(CMAKE_DISABLE_FIND_PACKAGE_GTest TRUE)
elseif(NOT CPPJIEBA_TEST_GTEST_PROVIDER STREQUAL "installed")
    message(FATAL_ERROR "Unknown GoogleTest provider: ${CPPJIEBA_TEST_GTEST_PROVIDER}")
endif()

include("${CPPJIEBA_SOURCE_DIR}/deps/googletest.cmake")
include("${CPPJIEBA_SOURCE_DIR}/deps/googletest.cmake")
if(NOT TARGET GTest::gtest)
    message(FATAL_ERROR "GoogleTest discovery did not provide GTest::gtest")
endif()
get_target_property(provider GTest::gtest CPPJIEBA_TEST_PROVIDER)
if(NOT provider STREQUAL CPPJIEBA_TEST_GTEST_PROVIDER)
    message(FATAL_ERROR "Expected ${CPPJIEBA_TEST_GTEST_PROVIDER}, selected ${provider}")
endif()
if(NOT provider STREQUAL "source" AND EXISTS "${CMAKE_CURRENT_BINARY_DIR}/_deps")
    message(FATAL_ERROR "A local GoogleTest provider should not populate FetchContent")
endif()

add_library(consumer INTERFACE)
target_link_libraries(consumer INTERFACE GTest::gtest)
